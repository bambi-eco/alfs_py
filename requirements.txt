# Runtime dependencies. Versions are intentionally loose so the package installs into as
# many existing environments as possible -- see docs/MIGRATION_REVIEW.md section 5.
#
# torch is NOT pinned. The rasteriser depends only on long-stable tensor primitives plus
# Tensor.scatter_reduce_ (>= 1.13), and alfspy.core.torchgl.compat probes that at import and
# falls back rather than breaking. Install the CPU or CUDA build that suits your machine,
# e.g. for CPU-only:
#   pip install torch --index-url https://download.pytorch.org/whl/cpu
torch>=1.13

# The ModernGL backend. Importing it is harmless without a GL driver -- a context is only
# created when that backend is actually selected -- so it is safe to install everywhere.
moderngl>=5.8.2

gltflib>=1.0.13
numpy>=1.23.5
opencv-python>=4.7.0.72
Pillow>=9.5.0
pyrr>=0.10.3
# The floor is 4.0, not 3.21: trimesh 3.x looks for the long-obsolete `pyembree` module and
# therefore silently falls back to the pure-Python ray intersector even when embreex is
# installed. Measured on a 131k-triangle DEM, that fallback is ~85x slower.
trimesh[easy]>=4.0,<6

# Optional accelerated ray/mesh backends for trimesh (label projection only). Pure-Python
# fallbacks are used when these are missing, so they are safe to drop on unsupported
# platforms.
Rtree>=1.0
embreex>=2.17; platform_machine == "x86_64" or platform_machine == "AMD64"
